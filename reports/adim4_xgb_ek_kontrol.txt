# Ek kontrol (seçim kuralının dışında): XGBoost ızgara kenarını genişletince HGB'yi geçiyor mu?
import pandas as pd
from arac_fiyat.compare import build_model
from arac_fiyat.train import CLEAN_CSV, run_cv, summarize_cv
from arac_fiyat.split import holdout_split
train, _ = holdout_split(pd.read_csv(CLEAN_CSV))
models = {f"xgb_d{d}_n{n}": build_model("xgb", {"max_depth": d, "n_estimators": n}) for d, n in [(10, 1500), (8, 3000), (10, 3000)]}
log = []
t = summarize_cv(run_cv(train, log, use_filter=False, models=models))
print(t.loc["doğrulama", ["MAPE", "MAPE_std", "medyan_APE"]].round(3).to_string())
print(t.loc["train", ["MAPE"]].round(2).to_string())
